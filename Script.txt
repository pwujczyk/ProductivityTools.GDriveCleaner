function myFunction() {
  var startTime = Date.now();
  var maxExecutionTimeMs = 5 * 60 * 1000; // 5 minutes (Apps Script limit is 6 minutes)
  var stopCounter = 0;
  var currentUser = Session.getActiveUser().getEmail();
  var files = DriveApp.searchFiles('trashed = false and "me" in owners and title contains "ToTrash"');
  while (files.hasNext()) {
    stopCounter++;
    if (stopCounter > 10000 || Date.now() - startTime > maxExecutionTimeMs) {
      console.log("Stopping early to avoid exceeding maximum execution time.");
      return;
    }
    var file = files.next();
    var name = file.getName();
    //console.log(name);
    if (name.startsWith("ToTrash")) {
      var ownerObject = file.getOwner();
      if (ownerObject != null) {
        var owner = ownerObject.getEmail();
        if (owner == currentUser) {
          console.log(owner);
          console.log(name);
          file.setTrashed(true);
        }
      }
    }
  }
}
